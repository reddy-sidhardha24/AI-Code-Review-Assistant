import java.util.*;

public class Main {
    public static void main(String[] args) {
        List<Integer> numbers = new ArrayList<>();

        for (int i = 0; i < 1000000; i++) {
            numbers.add(i);
        }

        for (Integer number : numbers) {
            if (numbers.contains(number)) {
                System.out.println(number);
            }
        }
    }
}
